async function analyzeFoodWithAI() { const apiKey = document.getElementById('apiKeyInput').value.trim(); const foodName = document.getElementById('foodNameInput').value.trim(); const foodGram = document.getElementById('foodGramInput').value.trim(); const imageFile = document.getElementById('imageInput').files[0]; const btn = document.getElementById('analyzeBtn'); if (!apiKey) { alert("AI ilə hesablamaq üçün yuxarıdakı xanaya Google Gemini API açarınızı yazmalısınız."); return; } if (!foodName && !imageFile) { alert("Zəhmət olmasa yeməyin adını və ya şəklini daxil edin."); return; } btn.disabled = true; btn.innerHTML = ` Gemini Analiz Edir...`; try { let parts = []; let prompt = `Yemək: ${foodName || 'Şəkildəki yemək'}. Məqsəd: ${foodGram ? foodGram + ' qram' : 'Standart porsiya'} üçün kalori və makroları dəqiq təxmin et. Cavabı yalnız aşağıdakı dəqiq JSON formatında qaytar:\n{"food": "${foodName || 'Yemək'}", "gram": ${foodGram || 100}, "calories": 250, "protein": 30, "carbs": 20, "fat": 5}`; parts.push({ text: prompt }); if (imageFile) { const base64Data = await fileToBase64(imageFile); parts.push({ inline_data: { mime_type: imageFile.type, data: base64Data.split(',')[1] } }); } // Ən son stabil model URL-i (v1 və ya gemini-2.5-flash) const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: parts }], generationConfig: { response_mime_type: "application/json" } }) }); const data = await response.json(); if (!response.ok) { // Əgər gemini-2.5-flash xətası verərsə, alternativ kimi gemini-1.5-flash-latest yoxlanılır const fallbackResponse = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash-latest:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: parts }], generationConfig: { response_mime_type: "application/json" } }) }); const fallbackData = await fallbackResponse.json(); if (!fallbackResponse.ok) { throw new Error(data.error ? data.error.message : "API Sorğu xətası"); } data.candidates = fallbackData.candidates; } let rawText = data.candidates[0].content.parts[0].text.trim(); const parsed = JSON.parse(rawText); tempAiParsed = parsed; document.getElementById('aiResultDetails').innerHTML = `

Yemək: ${parsed.food} (${parsed.gram}q)

Kalori: ${parsed.calories} kcal

Zülal: ${parsed.protein}g | Karbohidrat: ${parsed.carbs || 0}g | Yağ: ${parsed.fat || 0}g

`; document.getElementById('aiResultCard').classList.remove('hidden'); } catch (err) { console.error(err); alert("Xəta baş verdi: " + err.message); } finally { btn.disabled = false; btn.innerHTML = `AI ilə Avtomatik Hesabla `; } }
